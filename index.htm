    <script src="https://cdn.jsdelivr.net/npm/@tensorflow/tfjs@latest/dist/tf.min.js"></script>

    <script>

        

        

        const URL = "https://teachablemachine.withgoogle.com/models/S4-W6AOEH/";

        

        const CLASS_IDLE = “idle”;       // 平常沒動作的名稱

        const CLASS_LEFT = “left”;       // 左手正手掌檔的名稱

        const CLASS_RIGHT = “right”;     // 右手手背檔的名稱

        const CLASS_DOUBLE = “both”;   // 雙手握拳的名稱


        // ==========================================


        let model, webcam, labelContainer, maxPredictions;

        let isGameRunning = false;

        let hp = 100;

        let score = 0;

        let currentAttackType = "none";

        let hitCheckTimer = null;


        // 初始化與載入模型

        async function init() {

            document.getElementById("start-btn").style.display = "none";

            document.getElementById("status-message").innerText = "正在載入 AI 模型...";


            const modelURL = URL + "model.json";

            const metadataURL = URL + "metadata.json";


            try {

                // 載入對應函式庫與模型[cite: 2]

                model = await tmImage.load(modelURL, metadataURL);

                maxPredictions = model.getTotalClasses();


                const flip = true; 

                webcam = new tmImage.Webcam(640, 480, flip);

                await webcam.setup(); 

                await webcam.play();

                

                document.getElementById("webcam-container").appendChild(webcam.canvas);

                

                labelContainer = document.getElementById("label-container");

                labelContainer.innerHTML = "";

                for (let i = 0; i < maxPredictions; i++) {

                    let div = document.createElement("div");

                    div.className = "label-item";

                    labelContainer.appendChild(div);

                }


                document.getElementById("status-message").innerText = "系統就緒！3 秒後開始攻擊...";

                setTimeout(startGame, 3000);

                window.requestAnimationFrame(loop);


            } catch (error) {

                console.error(error);

                document.getElementById("status-message").innerText = "載入失敗，請確認鏡頭權限！";

                document.getElementById("start-btn").style.display = "inline-block";

            }

        }


        // 預測迴圈[cite: 2]

        async function loop() {

            webcam.update();

            await predict();

            if (hp > 0) {

                window.requestAnimationFrame(loop);

            }

        }


        // 進行預測與互動邏輯

        async function predict() {

            // 將畫面丟進模型預測[cite: 2]

            const prediction = await model.predict(webcam.canvas);

            

            let currentPose = "無偵測";

            let highestProb = 0;


            for (let i = 0; i < maxPredictions; i++) {

                const className = prediction[i].className;

                const probability = prediction[i].probability;

                

                labelContainer.childNodes[i].innerHTML = `${className}: ${(probability * 100).toFixed(0)}%`;


                if (probability > highestProb) {

                    highestProb = probability;

                    currentPose = className;

                }

            }


            // 依辨識結果觸發畫面變化 (大於 0.8 才判定成功)

            if (isGameRunning && highestProb > 0.8) {

                if (currentAttackType === "left" && currentPose === CLASS_LEFT) {

                    defendSuccess("🛡️ 左手掌格擋成功！+10分");

                } 

                else if (currentAttackType === "right" && currentPose === CLASS_RIGHT) {

                    defendSuccess("🛡️ 右手背格擋成功！+10分");

                }

                else if (currentAttackType === "double" && currentPose === CLASS_DOUBLE) {

                    defendSuccess("💎 雙手握拳防禦！+15分");

                }

            }

        }


        function startGame() {

            isGameRunning = true;

            hp = 100;

            score = 0;

            document.getElementById("status-message").innerText = "";

            updateStats();

            spawnAttack();

        }


        function spawnAttack() {

            if (hp <= 0) return;

            const rand = Math.floor(Math.random() * 3);

            let attackElement = null;


            if (rand === 0) {

                currentAttackType = "left";

                attackElement = document.getElementById("attack-left");

            } else if (rand === 1) {

                currentAttackType = "right";

                attackElement = document.getElementById("attack-right");

            } else {

                currentAttackType = "double";

                attackElement = document.getElementById("attack-double");

            }


            attackElement.style.display = "block";

            

            hitCheckTimer = setTimeout(() => {

                if (currentAttackType !== "none") {

                    takeDamage();

                    attackElement.style.display = "none";

                }

            }, 1500);

        }


        function defendSuccess(msg) {

            clearTimeout(hitCheckTimer);

            currentAttackType = "none";

            

            document.querySelectorAll('.attack').forEach(el => el.style.display = 'none');

            

            if (msg.includes("雙手")) {

                score += 15;

            } else {

                score += 10;

            }

            updateStats();


            document.getElementById("status-message").innerText = msg;

            

            // 觸發畫面變化：換背景色

            document.body.classList.add('flash-green');

            setTimeout(() => document.body.classList.remove('flash-green'), 250);


            setTimeout(() => {

                document.getElementById("status-message").innerText = "";

                spawnAttack();

            }, 1000);

        }


        function takeDamage() {

            currentAttackType = "none";

            hp -= 20;

            updateStats();

            

            document.getElementById("status-message").innerText = "💥 沒擋住！受到傷害 -20";

            

            // 觸發畫面變化：換背景色

            document.body.classList.add('flash-red');

            setTimeout(() => document.body.classList.remove('flash-red'), 250);


            if (hp <= 0) {

                gameOver();

            } else {

                setTimeout(() => {

                    document.getElementById("status-message").innerText = "";

                    spawnAttack();

                }, 1500);

            }

        }


        function updateStats() {

            document.getElementById("hp-display").innerText = hp;

            document.getElementById("score-display").innerText = score;

        }


        function gameOver() {

            isGameRunning = false;

            document.getElementById("status-message").innerText = `💀 護盾崩潰！\n最終分數：${score} 分`;

            

            const btn = document.getElementById("start-btn");

            btn.style.display = "inline-block";

            btn.innerText = "重新挑戰 (Restart)";

        }
